### #3 — invertirPalabras

<details>
<summary><strong>❌ Código original → ✅ Corrección</strong></summary>

```js
// ❌ Pau:
function invertirPalabras(str) {
  return str.split("").reverse().join("");
  // split("") divide por caracter: "hola mundo" → ["h","o","l","a"," ","m","u","n","d","o"]
}

// ✅ Debe ser:
function invertirPalabras(str) {
  return str.split(" ").reverse().join(" ");
  // split(" ") divide por palabras: "hola mundo" → ["hola", "mundo"]
}
```

</details>

🧠 **Patrón split-reverse-join**: mismo patrón que `esPalindromo` (`split("")`) pero con separador `" "`. Cuando querés palabras, dividís por espacio.

### #4 — masFrecuente

<details>
<summary><strong>❌ Código original → ✅ Corrección</strong></summary>

```js
// ❌ Pau (1er intento):
function masFrecuente(arr) {
  for (let i = 0; i < arr.length; i++) {
    if (arr.includes(arr[i])) {  // includes SIEMPRE es true — el elemento está en su propio array
      // ... no cuenta nada
    }
  }
  // Sin return → undefined
}

// ✅ Debe ser (Frequency Counter):
function masFrecuente(arr) {
  let maxCont = 0;
  let maxElem = arr[0];
  
  for (let i = 0; i < arr.length; i++) {
    let cont = 0;
    for (let j = 0; j < arr.length; j++) {
      if (arr[i] === arr[j]) cont++;
    }
    if (cont > maxCont) {
      maxCont = cont;
      maxElem = arr[i];
    }
  }
  return maxElem;
}
```

</details>

🧠 **Frequency Counter**: necesitás CONTAR cuántas veces aparece cada elemento, no preguntar si existe. Doble for + contador + variable de récord.

### #5 — primosHasta

```js
// ❌ Pau:
function primosHasta(n) {
  for (let i = 0; i < n; i++) {
    if (numero[i]) { ... }    // ❌ numero[i] — tratás n como array
  }
  for (let i = 0; i < n; i++) {
    // ...
  } else { ... }              // ❌ for...else no existe en JS

// ✅ Debe ser (Flag + break):
function primosHasta(n) {
  let primos = [];
  for (let i = 2; i <= n; i++) {
    let esPrimo = true;
    for (let j = 2; j < i; j++) {
      if (i % j === 0) {
        esPrimo = false;
        break;                // 🚫 encontró divisor, no es primo
      }
    }
    if (esPrimo) primos.push(i);
  }
  return primos;
}
```

</details>

🧠 **Flag Pattern**: variable booleana que se marca en el for interno si encontrás un divisor, y se consulta AL SALIR del for para decidir si agregás.

### #6 — estaOrdenado (versión #21)

<details>
<summary><strong>❌ Código original → ✅ Corrección</strong></summary>

```js
// ❌ Pau (versión #21 — nueva, mala):
function estaOrdenado(arr) {
  for (let i = 0; i < arr.length - 1; i++) {
    if (arr[i] < arr[i + 1]) {
      return true;    // ❌ return dentro del for: corta al primer par
    } else {
      return false;   // ❌ mismo problema, y si el array tiene 3+ elementos?
    }
  }
}

// ✅ Versión correcta (la del #12 que Pau ya tenía bien):
function estaOrdenado(arr) {
  for (let i = 0; i < arr.length - 1; i++) {
    if (arr[i] > arr[i + 1]) return false;  // 🚫 si encontrás un par roto → no está ordenado
  }
  return true;  // ✅ si terminaste el loop sin encontrar pares rotos → está ordenado
}
```

</details>

🧠 **Early return + recorrido completo**: si encontrás un par roto devolvés `false` inmediatamente. Si terminás el loop sin devolver, significa que todo está bien → `true`.

### E1 — validarEmail

<details>
<summary><strong>❌ Código original → ✅ Corrección</strong></summary>

```js
// ❌ Pau:
function validarEmail(email) {
    let condicion = false;
    let array = email.toLowerCase().split("");
    for (let i = 0; i <= array.length; i++) {
        if (array[i] === "@") {
            for (let j = array[i]; j <= array.length; j++) {
                if (array[j] === "") condicion = true;
            }
        }
    }
    return condicion;
}

// ✅ Debe ser:
function validarEmail(email) {
    let arroba = email.indexOf("@");
    if (arroba === -1) return false;
    let dominio = email.slice(arroba + 1);
    return dominio.includes(".");
}
```

</details>

🧠 **Patrón indexOf + slice + includes**: para verificar estructura de email no hace falta for anidado. Buscá posición del `@`, verificá que exista, agarrá lo que sigue, preguntá si tiene punto.

### E2 — generarSlug

```js
// ❌ Pau:

    return frase.trim();

// ✅ Debe ser:
function generarSlug(texto) {
    let frase = texto.toLowerCase().trim().split(" ");
    let resultado = [];
    for (let i = 0; i < frase.length; i++) {
        if (frase[i] !== "") resultado.push(frase[i]);
    }
    return resultado.join("-");
}
```

</details>

### E8 — ordenarPorPrecio

<details>
<summary><strong>❌ Código original → ✅ Corrección</strong></summary>

```js
// ❌ Pau:
function ordenarPorPrecio(productos) {
    let temporal = 0;
    for (let j = 0; j < productos.length; j++) {
        if (productos[j].precio > productos[j + 1].precio) {
            temporal = productos[j];
            productos[j + 1] = productos[j];  // B se pierde
            productos[j] = temporal;          // A vuelve a donde estaba
        }
    }
}

// ✅ Debe ser (swap correcto):
function ordenarPorPrecio(productos) {
    let temporal;
    for (let i = 0; i < productos.length; i++) {
        for (let j = 0; j < productos.length - 1; j++) {
            if (productos[j].precio > productos[j + 1].precio) {
                temporal = productos[j];
                productos[j] = productos[j + 1];
                productos[j + 1] = temporal;
            }
        }
    }
    return productos;
}
```

</details>

🧠 **Bubble sort + swap correcto**: guardar A en temporal, mover B a posición de A, recuperar A desde temporal a posición de B. El bucle interno va hasta `length - 1` para no acceder a `undefined`.
