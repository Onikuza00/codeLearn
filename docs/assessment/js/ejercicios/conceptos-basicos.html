1 🟢 sumarPares(n) 🔁 Repasar

Suma todos los números pares desde 1 hasta n (inclusive).

sumarPares(6) → 12  // 2+4+6
sumarPares(10) → 30  // 2+4+6+8+10
sumarPares(1) → 0 
function sumarPares(n) {
  let suma = 0;
  for (let i = 2; i <= n; i += 2) suma += i;
  return suma;
}
2 🟢 contarPalabras(str) 🔁 Repasar

Recibe una frase y devuelve cuántas palabras tiene separadas por espacios.

contarPalabras("Hola mundo") → 2
contarPalabras("JS mola mucho") → 3
contarPalabras("") → 0 
function contarPalabras(str) {
  if (str.length === 0) return 0;
  let espacios = 0;
  for (let i = 0; i < str.length; i++) {
    if (str[i] === " ") espacios++;
  }
  return espacios + 1;
}
3 🟡 eliminarDuplicados(arr) ✅ Completado

Recibe un array y devuelve uno nuevo sin elementos repetidos, manteniendo el orden original.

eliminarDuplicados([1, 2, 2, 3, 4, 4, 5]) → [1, 2, 3, 4, 5]
eliminarDuplicados(["a", "a", "b"]) → ["a", "b"] 
function eliminarDuplicados(arr) {
  let resultado = [];
  for (let i = 0; i < arr.length; i++) {
    if (!resultado.includes(arr[i])) resultado.push(arr[i]);
  }
  return resultado;
}
4 🟡 stringMasLargo(arr) ✅ Completado

Recibe un array de strings y devuelve el que tenga más caracteres.

stringMasLargo(["sol", "ventana", "mar"]) → "ventana"
stringMasLargo(["js", "css", "html"]) → "html" 
function stringMasLargo(arr) {
  let mayor = arr[0];
  for (let i = 1; i < arr.length; i++) {
    if (arr[i].length > mayor.length) mayor = arr[i];
  }
  return mayor;
}
5 🟡 esPalindromo(str) 🔁 Repasar

¿La frase se lee igual al derecho y al revés? Ignorar mayúsculas no es necesario.

esPalindromo("radar") → true
esPalindromo("hola") → false
esPalindromo("somos") → true

💡 no existe .reverse() en strings — convertí a array con .split("")

function esPalindromo(str) {
  return str.split("").reverse().join("") === str;
}
6 🟡 filtrarMayores(arr, n) 🔁 Repasar

Devuelve un nuevo array solo con los números mayores que n.

filtrarMayores([3, 8, 1, 12], 5) → [8, 12]
filtrarMayores([1, 2, 3], 10) → []

💡 compará item > n, no item.length — los números no tienen .length

function filtrarMayores(arr, n) {
  let resultado = [];
  for (let i = 0; i < arr.length; i++) {
    if (arr[i] > n) resultado.push(arr[i]);
  }
  return resultado;
}
7 🔴 contarLetra(str, letra) ✅ Completado

Cuenta cuántas veces aparece una letra en un string, sin distinguir mayúsculas/minúsculas.

contarLetra("JavaScript", "a") → 2
contarLetra("Cascara", "c") → 3  // C y c cuentan
contarLetra("Hola", "z") → 0 
function contarLetra(str, letra) {
  let count = 0;
  for (let i = 0; i < str.length; i++) {
    if (str[i].toLowerCase() === letra.toLowerCase()) count++;
  }
  return count;
}
8 🔴 promedio(arr) ✅ Completado

Calcula el promedio de todos los números del array. Si está vacío devuelve 0.

promedio([10, 20, 30]) → 20
promedio([5, 5, 5, 5]) → 5
promedio([]) → 0 
function promedio(arr) {
  if (arr.length === 0) return 0;
  let suma = 0;
  for (let i = 0; i < arr.length; i++) suma += arr[i];
  return suma / arr.length;
}
9 🟡 invertirElOrden(str) ✅ Completado

Invierte el orden de los caracteres en un string.

invertirElOrden("hola") → "aloh"
invertirElOrden("JS") → "SJ" 
function invertirElOrden(str) {
  return str.split("").reverse().join("");
}
10 🟡 contarMayusculas(str) ✅ Completado

Cuenta cuántas letras mayúsculas hay en el string.

contarMayusculas("Hola Mundo") → 2  // H, M
contarMayusculas("javascript") → 0
contarMayusculas("JS") → 2 
function contarMayusculas(str) {
  let count = 0;
  for (let i = 0; i < str.length; i++) {
    if (str[i] !== str[i].toLowerCase()) count++;
  }
  return count;
}
11 🔴 numerosImpares(n) 🔁 Repasar

Devuelve un array con todos los números impares desde 1 hasta n.

numerosImpares(5) → [1, 3, 5]

numerosImpares(1) → [1]

12 🔴 estaOrdenado(arr) 🔁 Repasar

¿Los números están ordenados de menor a mayor?

estaOrdenado([1, 3, 5]) → true
estaOrdenado([3, 1, 5]) → false
estaOrdenado([1, 2, 2, 3]) → true 
function estaOrdenado(arr) {
  for (let i = 0; i < arr.length - 1; i++) {
    if (arr[i] > arr[i + 1]) return false;
  }
  return true;
}

function sumaArray(arr) {

}
14 🔴 esPrimo(n) 🔁 Repasar

¿El número es primo? Solo divisible por 1 y por sí mismo.

esPrimo(7) → true
esPrimo(4) → false  // divisible por 2
esPrimo(1) → false  // 1 no es primo 
function esPrimo(n) {
  if (n < 2) return false;
  for (let i = 2; i < n; i++) {
    if (n % i === 0) return false;
  }
  return true;
}
15 🟡 limpiarEspacios(str) ⏳ Pendiente

Elimina los espacios múltiples consecutivos dejando solo uno entre palabras.

limpiarEspacios("hola mundo") → "hola mundo"
limpiarEspacios("js es genial") → "js es genial" 
function limpiarEspacios(str) {
  let resultado = "";
  for (let i = 0; i < str.length; i++) {
    if (str[i] !== " " || str[i + 1] !== " ") resultado += str[i];
  }
  return resultado;
}
