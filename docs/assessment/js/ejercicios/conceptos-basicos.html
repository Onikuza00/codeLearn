1 🟢 sumarPares(n) 🔁 Repasar

Suma todos los números pares desde 1 hasta n (inclusive).

sumarPares(6) → 12  // 2+4+6
sumarPares(10) → 30  // 2+4+6+8+10
sumarPares(1) → 0 
function sumarPares(n) {
  let suma = 0;
  for (let i = 2; i <= n; i += 2) suma += i;
  return suma;
}
2 🟢 contarPalabras(str) 🔁 Repasar

Recibe una frase y devuelve cuántas palabras tiene separadas por espacios.

contarPalabras("Hola mundo") → 2
contarPalabras("JS mola mucho") → 3
contarPalabras("") → 0 
function contarPalabras(str) {
  if (str.length === 0) return 0;
  let espacios = 0;
  for (let i = 0; i < str.length; i++) {
    if (str[i] === " ") espacios++;
  }
  return espacios + 1;
}
3 🟡 eliminarDuplicados(arr) ✅ Completado

Recibe un array y devuelve uno nuevo sin elementos repetidos, manteniendo el orden original.

eliminarDuplicados([1, 2, 2, 3, 4, 4, 5]) → [1, 2, 3, 4, 5]
eliminarDuplicados(["a", "a", "b"]) → ["a", "b"] 
function eliminarDuplicados(arr) {
  let resultado = [];
  for (let i = 0; i < arr.length; i++) {
    if (!resultado.includes(arr[i])) resultado.push(arr[i]);
  }
  return resultado;
}
4 🟡 stringMasLargo(arr) ✅ Completado

Recibe un array de strings y devuelve el que tenga más caracteres.

stringMasLargo(["sol", "ventana", "mar"]) → "ventana"
stringMasLargo(["js", "css", "html"]) → "html" 
function stringMasLargo(arr) {
  let mayor = arr[0];
  for (let i = 1; i < arr.length; i++) {
    if (arr[i].length > mayor.length) mayor = arr[i];
  }
  return mayor;
}

7 🔴 contarLetra(str, letra) ✅ Completado

Cuenta cuántas veces aparece una letra en un string, sin distinguir mayúsculas/minúsculas.

function contarLetra(str, letra) {
  let count = 0;
  for (let i = 0; i < str.length; i++) {
    if (str[i].toLowerCase() === letra.toLowerCase()) count++;

8 🔴 promedio(arr) ✅ Completado

Calcula el promedio de todos los números del array. Si está vacío devuelve 0.

promedio([10, 20, 30]) → 20
promedio([5, 5, 5, 5]) → 5
promedio([]) → 0 
function promedio(arr) {
  if (arr.length === 0) return 0;
  let suma = 0;
  for (let i = 0; i < arr.length; i++) suma += arr[i];
  return suma / arr.length;
}

14 🔴 esPrimo(n) 🔁 Repasar

¿El número es primo? Solo divisible por 1 y por sí mismo.

esPrimo(7) → true
esPrimo(4) → false  // divisible por 2
esPrimo(1) → false  // 1 no es primo 
function esPrimo(n) {
  if (n < 2) return false;
  for (let i = 2; i < n; i++) {
    if (n % i === 0) return false;
  }
  return true;
}

11 🟡 agruparPorLetra(palabras) ✅ Completado

Array de strings → objeto agrupado por primera letra. Cada clave es una letra, cada valor un array de palabras.

agruparPorLetra(["hola", "mundo", "hielo", "moto"]) → { h: ["hola", "hielo"], m: ["mundo", "moto"] } 
function agruparPorLetra(palabras) {
  let grupos = {};
  for (let i = 0; i < palabras.length; i++) {
    let letra = palabras[i][0];
    if (grupos[letra]) {
      grupos[letra].push(palabras[i]);
    } else {
      grupos[letra] = [palabras[i]];
    }
  }
  return grupos;
}
12 🟡 mediaAprobados(estudiantes) ✅ Completado

Array de { nombre, nota } → media de los que aprobaron (nota >= 5). Si nadie aprueba → null.

mediaAprobados([{nombre:"Ana", nota:7}, {nombre:"Luis", nota:4}, {nombre:"Pau", nota:6}]) → 6.5
mediaAprobados([{nombre:"Luis", nota:4}]) → null 
function mediaAprobados(estudiantes) {
  let suma = 0, count = 0;
  for (let i = 0; i < estudiantes.length; i++) {
    if (estudiantes[i].nota >= 5) {
      suma += estudiantes[i].nota;
      count++;
    }
  }
  return count === 0 ? null : suma / count;
}
13 🟡 eliminarDuplicados(arr) ✅ Completado

function eliminarDuplicados(arr) {

  for (let i = 0; i < arr.length; i++) {

14 🟡 sumarMatriz(matriz) ✅ Completado

Array de arrays de números → suma total de todos los números. Doble for.

sumarMatriz([[1, 2], [3, 4, 5], [6]]) → 21
sumarMatriz([[1], []]) → 1 
function sumarMatriz(matriz) {
  let total = 0;
  for (let i = 0; i < matriz.length; i++) {
    for (let j = 0; j < matriz[i].length; j++) {
      total += matriz[i][j];
    }
  }
  return total;
}

1 🟢 extraerDominio(email) ⏳ Pendiente

Devuelve solo el dominio de un email (lo que va después del @). Si no tiene @ o está vacío, devuelve null.

extraerDominio("pau@gmail.com") → "gmail.com"
extraerDominio("hola@empresa.co") → "empresa.co"
extraerDominio("sinarroba") → null 
function extraerDominio(email) {
  if (typeof email !== "string" || email.length === 0) return null;
  let pos = email.indexOf("@");
  if (pos === -1) return null;
  return email.slice(pos + 1);
}
2 🟡 esSlugValido(slug) ⏳ Pendiente

¿El slug solo tiene minúsculas, números y guiones? Sin espacios, mayúsculas ni especiales. Sin regex.

esSlugValido("mi-articulo-1") → true
esSlugValido("Mi Articulo") → false
esSlugValido("") → false 
function esSlugValido(slug) {
  if (slug.length === 0) return false;
  for (let i = 0; i < slug.length; i++) {
    let c = slug[i];
    if (c >= "a" && c <= "z") continue;
    if (c >= "0" && c <= "9") continue;
    if (c === "-") continue;
    return false;
  }
  return true;
}
3 🟢 formatearFecha(fecha) ⏳ Pendiente

Recibe "AAAA-MM-DD" → devuelve "DD de mes del AAAA". Usá array de meses.

formatearFecha("2026-07-28") → "28 de julio del 2026"
formatearFecha("") → null 
function formatearFecha(fecha) {
  if (!fecha || fecha.length === 0) return null;
  const meses = ["enero", "febrero", "marzo", "abril", "mayo", "junio", "julio", "agosto", "septiembre", "octubre", "noviembre", "diciembre"];
  let partes = fecha.split("-");
  return partes[2] + " de " + meses[partes[1] - 1] + " del " + partes[0];
}
4 🟡 productoMasCaro(productos) ⏳ Pendiente

Array de { nombre, precio } → el nombre del más caro. Sin Math.max. Vacío → null.

productoMasCaro([{nombre:"Mouse", precio:25}, {nombre:"Monitor", precio:250}]) → "Monitor"
productoMasCaro([]) → null 
function productoMasCaro(productos) {
  if (productos.length === 0) return null;
  let maxPrecio = productos[0].precio;
  let maxNombre = productos[0].nombre;
  for (let i = 1; i < productos.length; i++) {
    if (productos[i].precio > maxPrecio) {
      maxPrecio = productos[i].precio;
      maxNombre = productos[i].nombre;
    }
  }
  return maxNombre;
}
5 🟡 esContrasenaSegura(pwd) ⏳ Pendiente

¿Cumple todas? 8+ caracteres, 1 mayúscula, 1 número, 1 especial (!@#$%).

esContrasenaSegura("Hola123!") → true
esContrasenaSegura("hola123!") → false
esContrasenaSegura("HOLA1234") → false 
function esContrasenaSegura(pwd) {
  if (pwd.length < 8) return false;
  let mayus = false, num = false, special = false;
  const especiales = "!@#$%";
  for (let i = 0; i < pwd.length; i++) {
    let c = pwd[i];
    if (c >= "A" && c <= "Z") mayus = true;
    if (c >= "0" && c <= "9") num = true;
    if (especiales.includes(c)) special = true;
  }
  return mayus && num && special;
}

7 🟡 totalCarrito(items) ⏳ Pendiente

Array de { nombre, precio, cantidad } → total. Si supera 100€, 10% descuento. Vacío → 0.

totalCarrito([{nombre:"Mouse", precio:25, cantidad:2}, {nombre:"Teclado", precio:80, cantidad:1}]) → 117
totalCarrito([]) → 0 
function totalCarrito(items) {
  if (items.length === 0) return 0;
  let total = 0;
  for (let i = 0; i < items.length; i++) {
    total += items[i].precio * items[i].cantidad;
  }
  if (total > 100) total -= total * 10 / 100;
  return total;
}
8 🟡 limpiarEspacios(texto) ⏳ Pendiente

Elimina espacios múltiples dejando solo uno entre palabras. Sin trim(), sin regex, sin replace().

limpiarEspacios(" hola mundo ") → "hola mundo"
limpiarEspacios("") → "" 
function limpiarEspacios(texto) {
  let res = "";
  for (let i = 0; i < texto.length; i++) {
    if (texto[i] === " " && texto[i - 1] === " ") continue;
    res += texto[i];
  }
  if (res[0] === " ") res = res.slice(1);
  if (res[res.length - 1] === " ") res = res.slice(0, -1);
  return res;
}
9 🔴 frecuenciaCaracteres(texto) ✅ Completado

Objeto con la frecuencia de cada caracter (solo letras, sin espacios).

frecuenciaCaracteres("hola") → { h: 1, o: 1, l: 1, a: 1 }
frecuenciaCaracteres("javascript") → { j: 1, a: 2, v: 1, s: 1, c: 1, r: 1, i: 1, p: 1, t: 1 } 
function frecuenciaCaracteres(texto) {
  let freq = {};
  for (let i = 0; i < texto.length; i++) {
    let c = texto[i];
    if (c === " ") continue;
    if (freq[c]) freq[c]++;
    else freq[c] = 1;
  }
  return freq;
}
10 🔴 ordenarPorPrecio(productos) ✅ Completado

Bubble sort. Ordená { nombre, precio } de menor a mayor. Sin .sort(). Sin mutar original.

ordenarPorPrecio([{nombre:"Monitor", precio:250}, {nombre:"Mouse", precio:25}]) → [{nombre:"Mouse", precio:25}, {nombre:"Monitor", precio:250}] 
function ordenarPorPrecio(productos) {
  let res = [...productos];
  for (let i = 0; i < res.length; i++) {
    for (let j = 0; j < res.length - 1; j++) {
      if (res[j].precio > res[j + 1].precio) {
        let temp = res[j];
        res[j] = res[j + 1];
        res[j + 1] = temp;
      }
    }
  }
  return res;
}
