Primera vez usando el EntityManager en este proyecto — hasta ahora solo leíste con el Repository. persist() solo prepara, flush() ejecuta el INSERT real. Ver Doctrine → El EntityManager.

#[Route('/tareas', name: 'tarea_crear', methods: ['POST'])]
public function crear(Request $request, EntityManagerInterface $em): Response
{
    $tarea = new Tarea();
    $tarea->setTitle($request->query->get('title'));
    $tarea->setDone(false);
    $tarea->setCreatedAt(new \DateTimeImmutable());

    $em->persist($tarea);
    $em->flush();

    return $this->redirectToRoute('tarea_index');
}

// src/Repository/TareaRepository.php
public function findPendientes(): array
{
    return $this->createQueryBuilder('t')
        ->where('t.done = :done')
        ->setParameter('done', false)
        ->orderBy('t.createdAt', 'DESC')
        ->getQuery()
        ->getResult();
}

// src/Service/FileUploadValidator.php — contrástalo con la doc 07-servicios
namespace App\Service;

use Symfony\Component\HttpFoundation\File\UploadedFile;

class FileUploadValidator
{
    private const MAX_BYTES = 2 * 1024 * 1024;
    private const ALLOWED = ['image/jpeg', 'image/png'];

    /** @return string[] */
    public function validate(UploadedFile $file): array
    {
        $errors = [];
        if ($file->getSize() > self::MAX_BYTES) {
            $errors[] = 'El archivo supera los 2 MB.';
        }
        if (!in_array($file->getMimeType(), self::ALLOWED, true)) {
            $errors[] = 'Formato no permitido (solo JPG o PNG).';
        }

        return $errors;
    }
}

#[Route('/product/{id}/image', name: 'product_image', methods: ['POST'])]
public function image(Request $request, Product $product, EntityManagerInterface $em, FileUploadValidator $validator): Response
{
    $file = $request->files->get('image');

    if ($errors) {

    $em->flush();

}
